      async function loadChart() {
        const sym = document.getElementById("csym").value.trim().toUpperCase();
        const days = document.getElementById("cdays").value;
        if (!sym) return;
        document.getElementById("spwrap").style.display = "inline";
        document.getElementById("clabel").textContent = sym + " - " + curTF;
        try {
          const d = await fj(
            "/api/chart4/" + sym + "?days=" + days + "&timeframe=" + curTF,
          );
          if (!d || !d.dates || !d.dates.length) throw new Error("No data");
          lastData = d;
          drawPrice(d);
          drawRSI(d);
          const cl = d.close,
            last = cl[cl.length - 1],
            prev = cl[cl.length - 2] || last;
          const chg = (((last - prev) / prev) * 100).toFixed(2);
          const col = chg >= 0 ? "#68d391" : "#fc8181";
          document.getElementById("pdisplay").innerHTML =
            "$" +
            last +
            ' <span style="color:' +
            col +
            ';font-size:11px">' +
            (chg >= 0 ? "+" : "") +
            chg +
            "%</span>";
          showTrend(d);
          buildLegend();
          chartLoaded = true;
        } catch (e) {
          document.getElementById("clabel").textContent = "Error: " + e.message;
        } finally {
          document.getElementById("spwrap").style.display = "none";
        }

      function showTrend(d) {
        const cl = d.close.filter((v) => v != null);
        if (cl.length < 10) return;
        const last = cl[cl.length - 1];
        const ma50 = d.ma50 ? d.ma50.filter((v) => v != null) : [];
        const ma200 = d.ma200 ? d.ma200.filter((v) => v != null) : [];
        const lma50 = ma50[ma50.length - 1],
          lma200 = ma200[ma200.length - 1];
        const chg20 =
          cl.length >= 20
            ? (
                ((last - cl[cl.length - 20]) / cl[cl.length - 20]) *
                100
              ).toFixed(1)
            : null;
        let trend = "tside",
          label = "Sideways";
        if (lma50 && lma200) {
          if (lma50 > lma200 && last > lma50) {
            trend = "tbull";
            label = "Bullish";
          } else if (lma50 < lma200 && last < lma50) {
            trend = "tbear";
            label = "Bearish";
          }
        } else if (chg20 != null) {
          if (chg20 > 3) {
            trend = "tbull";
            label = "Bullish";
          } else if (chg20 < -3) {
            trend = "tbear";
            label = "Bearish";
          }
        }
        const rsi = d.rsi ? d.rsi.filter((v) => v != null) : [];
        const lr = rsi[rsi.length - 1];
        let rl = "";
        if (lr != null) {
          if (lr > 70) rl = " · OB";
          else if (lr < 30) rl = " · OS";
        }
        document.getElementById("trbdg").innerHTML =
          '<span class="trend-badge ' + trend + '">' + label + rl + "</span>";
        if (lr != null)
          document.getElementById("rsiv").textContent = "RSI " + lr.toFixed(1);
      }

      function setupCross(canvas, data) {
        const tip = document.getElementById("xtip");
        function handle(mx) {
          const pad = { l: 52, r: 10 };
          const cw = canvas.width - pad.l - pad.r;
          const n = data.dates.length;
          const idx = Math.min(
            n - 1,
            Math.max(0, Math.round(((mx - pad.l) / cw) * (n - 1))),
          );
          crossX = idx;
          if (curCT === "candle") drawCandles(data);
          const rows = [
            '<div class="xhdr">' + data.dates[idx] + "</div>",
            '<div class="xrow"><span><span class="xdot" style="background:#63b3ed"></span>Price</span><span>$' +
              (data.close[idx] || "-") +
              "</span></div>",
          ];
          if (maVis.ma21 && data.ma21[idx])
            rows.push(
              '<div class="xrow"><span><span class="xdot" style="background:#f6e05e"></span>MA21</span><span>$' +
                data.ma21[idx] +
                "</span></div>",
            );
          if (maVis.ma50 && data.ma50[idx])
            rows.push(
              '<div class="xrow"><span><span class="xdot" style="background:#68d391"></span>MA50</span><span>$' +
                data.ma50[idx] +
                "</span></div>",
            );
          if (maVis.ma200 && data.ma200[idx])
            rows.push(
              '<div class="xrow"><span><span class="xdot" style="background:#fc8181"></span>MA200</span><span>$' +
                data.ma200[idx] +
                "</span></div>",
            );
          if (data.rsi && data.rsi[idx])
            rows.push(
              '<div class="xrow"><span><span class="xdot" style="background:#b794f4"></span>RSI</span><span>' +
                data.rsi[idx] +
                "</span></div>",
            );
          tip.innerHTML = rows.join("");
          tip.style.left =
            (mx + 140 > canvas.width ? mx - 145 : mx + 10) + "px";
          tip.style.top = "30px";
          tip.style.display = "block";
        }
        canvas.addEventListener("mousemove", (e) => {
          const rect = canvas.getBoundingClientRect();
          handle(e.clientX - rect.left);
        });
        canvas.addEventListener("mouseleave", () => {
          tip.style.display = "none";
          crossX = -1;
          if (curCT === "candle" && lastData) drawCandles(lastData);
        });
        canvas.addEventListener(
          "touchmove",
          (e) => {
            e.preventDefault();
            const rect = canvas.getBoundingClientRect();
            handle(e.touches[0].clientX - rect.left);
          },
          { passive: false },
        );
      }

      function drawCandles(d) {
        const ex = Chart.getChart("pc");
        if (ex) ex.destroy();
        const canvas = document.getElementById("pc");
        const ctx = canvas.getContext("2d");
        const W = canvas.parentElement.clientWidth - 24,
          H = 240;
        canvas.width = W;
        canvas.height = H;
        const n = d.dates.length;
        if (!n) return;
        const pad = { l: 52, r: 10, t: 8, b: 28 };
        const cw = W - pad.l - pad.r,
          ch = H - pad.t - pad.b;
        const prices = [...d.open, ...d.high, ...d.low, ...d.close].filter(
          (v) => v != null,
        );
        const minP = Math.min(...prices) * 0.998,
          maxP = Math.max(...prices) * 1.002;
        const toY = (v) => pad.t + ch - ((v - minP) / (maxP - minP)) * ch;
        const bw = Math.max(2, Math.floor((cw / n) * 0.7));
        const xOf = (i) => pad.l + i * (cw / (n - 1 || 1));
        ctx.clearRect(0, 0, W, H);
        ctx.font = "10px -apple-system,sans-serif";
        for (let i = 0; i <= 4; i++) {
          const y = pad.t + (ch / 4) * i;
          ctx.strokeStyle = "#0d1220";
          ctx.lineWidth = 0.5;
          ctx.beginPath();
          ctx.moveTo(pad.l, y);
          ctx.lineTo(W - pad.r, y);
          ctx.stroke();
          ctx.fillStyle = "#4a5568";
          ctx.textAlign = "right";
          ctx.fillText(
            "$" + (maxP - ((maxP - minP) / 4) * i).toFixed(0),
            pad.l - 3,
            y + 3,
          );
        }
        ctx.fillStyle = "#4a5568";
        ctx.textAlign = "center";
        const step = Math.max(1, Math.floor(n / 6));
        for (let i = 0; i < n; i += step)
          ctx.fillText(d.dates[i].substring(5, 10), xOf(i), H - 4);
        for (let i = 0; i < n; i++) {
          const o = d.open[i],
            h = d.high[i],
            l = d.low[i],
            c = d.close[i];
          if (o == null || c == null) continue;
          const x = xOf(i),
            up = c >= o;
          ctx.strokeStyle = up ? "#68d391" : "#fc8181";
          ctx.fillStyle = up ? "#68d391" : "#fc8181";
          ctx.lineWidth = 1;
          ctx.beginPath();
          ctx.moveTo(x, toY(h));
          ctx.lineTo(x, toY(l));
          ctx.stroke();
          const by = toY(Math.max(o, c)),
            bh = Math.max(1, Math.abs(toY(o) - toY(c)));
          ctx.fillRect(x - bw / 2, by, bw, bh);
        }
        [
          { k: "ma21", data: d.ma21, col: "#f6e05e" },
          { k: "ma50", data: d.ma50, col: "#68d391" },
          { k: "ma200", data: d.ma200, col: "#fc8181" },
        ].forEach(({ k, data, col }) => {
          if (!maVis[k] || !data) return;
          ctx.strokeStyle = col;
          ctx.lineWidth = 1.5;
          ctx.beginPath();
          let s = false;
          for (let i = 0; i < n; i++) {
            if (data[i] == null) continue;
            const x = xOf(i),
              y = toY(data[i]);
            if (!s) {
              ctx.moveTo(x, y);
              s = true;
            } else ctx.lineTo(x, y);
          }
          ctx.stroke();
        });
        if (crossX >= 0 && crossX < n) {
          const x = xOf(crossX);
          ctx.strokeStyle = "rgba(255,255,255,0.15)";
          ctx.lineWidth = 1;
          ctx.setLineDash([3, 3]);
          ctx.beginPath();
          ctx.moveTo(x, pad.t);
          ctx.lineTo(x, H - pad.b);
          ctx.stroke();
          ctx.setLineDash([]);
        }
        setupCross(canvas, d);
        pChart = null;
      }

      function getTB(r) {
        const v = r.total_return_pct;
        if (v > 5) return '<span class="trend-badge tbull">Profitable</span>';
        if (v < -5) return '<span class="trend-badge tbear">Losing</span>';
        return '<span class="trend-badge tside">Neutral</span>';
      }
      function getSTag(s) {
        if (s.includes("EMA") || s.includes("ema")) return "tema";
        if (s.includes("RSI")) return "trsi";
        return "tma";
      }

      async function runBT() {
        const sym = document.getElementById("bsym").value.trim().toUpperCase();
        const days = parseInt(document.getElementById("bdays").value);
        const cash = parseFloat(document.getElementById("bcash").value);
        if (!sym) {
          alert("Enter a symbol");
          return;
        }

        // Collect dynamic params
        const params = {
          symbol: sym,
          days,
          cash,
          strategy: selectedBTStrategy,
        };
        document.querySelectorAll('[id^="btp-"]').forEach((inp) => {
          const key = inp.id.replace("btp-", "");
          params[key] = parseFloat(inp.value) || inp.value;
        });
        // Defaults for standard strategies
        if (!params.fast_ma) params.fast_ma = 50;
        if (!params.slow_ma) params.slow_ma = 200;
        if (!params.rsi_period) params.rsi_period = 14;
        if (!params.oversold) params.oversold = 30;
        if (!params.overbought) params.overbought = 70;

        const btn = document.getElementById("rbtn");
        btn.disabled = true;
        btn.textContent = "Running...";
        dBC();
        document.getElementById("btout").innerHTML =
          '<div class="loading"><div class="spin" style="width:24px;height:24px;border-width:3px;margin-bottom:10px"></div><div>Running backtest on ' +
          sym +
          "...</div></div>";

        try {
          // Use strategy-specific endpoint for custom strategies
          const isCustom =
            selectedBTStrategy !== "both" &&
            selectedBTStrategy !== "ma_crossover" &&
            selectedBTStrategy !== "rsi_strategy";
          const endpoint = isCustom
            ? "/api/backtest/strategy"
            : "/api/backtest/run";
          const body = isCustom
            ? { ...params, strategy: selectedBTStrategy }
            : {
                symbol: sym,
                strategy: selectedBTStrategy,
                days,
                cash,
                fast_ma: params.fast_ma,
                slow_ma: params.slow_ma,
                rsi_period: params.rsi_period,
                oversold: params.oversold,
                overbought: params.overbought,
              };

          const res = await fetch(endpoint, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body),
          });
          const data = await res.json();
          if (!res.ok) throw new Error(data.detail || "Server error");
          renderBT(data.results, sym, cash);
        } catch (e) {
          document.getElementById("btout").innerHTML =
            '<div style="margin:14px;padding:12px;background:#2d0f0f;border:1px solid #742a2a;border-radius:9px;color:#fc8181">Error: ' +
            e.message +
            "</div>";
        } finally {
          btn.disabled = false;
          btn.textContent = "Run Backtest";
        }
      }

          "$" +
